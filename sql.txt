CREATE TABLE usuarios (
    id_usuario INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    cedula CHAR(8) NOT NULL UNIQUE,
    usuario VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    rol ENUM('docente', 'encargado') NOT NULL,
    activo BOOLEAN NOT NULL DEFAULT TRUE
) ENGINE = InnoDB;

CREATE TABLE prioridades (
    id_prioridad TINYINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20) NOT NULL UNIQUE
) ENGINE = InnoDB;

CREATE TABLE estados (
    id_estado TINYINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(30) NOT NULL UNIQUE
) ENGINE = InnoDB;

CREATE TABLE tickets (
    id_ticket INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,

    descripcion TEXT NOT NULL,

    fecha DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,

    sala ENUM(
        'Salón 5',
        'Salón 6',
        'Salón 7',
        'Adscripción',
        'Dirección',
        'Campus'
    ) NULL,

    id_usuario INT UNSIGNED NOT NULL,
    id_estado TINYINT UNSIGNED NOT NULL,
    id_prioridad TINYINT UNSIGNED NOT NULL,

    CONSTRAINT fk_ticket_usuario
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT fk_ticket_estado
        FOREIGN KEY (id_estado)
        REFERENCES estados(id_estado)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    CONSTRAINT fk_ticket_prioridad
        FOREIGN KEY (id_prioridad)
        REFERENCES prioridades(id_prioridad)
        ON DELETE RESTRICT
        ON UPDATE CASCADE,

    INDEX idx_ticket_fecha (fecha),
    INDEX idx_ticket_estado_fecha (id_estado, fecha),
    INDEX idx_ticket_prioridad_fecha (id_prioridad, fecha),
    INDEX idx_ticket_usuario_fecha (id_usuario, fecha)

) ENGINE = InnoDB;
